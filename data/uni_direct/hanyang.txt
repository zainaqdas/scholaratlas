Liferay.on('allPortletsReady', function() {
    // 1. 현재 페이지의 Plid를 가져옵니다. (숫자 비교를 위해 Number 타입으로 변환)
    var currentPlid = Number(Liferay.ThemeDisplay.getPlid());
    
    // 2. 메뉴 아이템 순회
    document.querySelectorAll('.topMenu-list .list-item').forEach(function(item) {
        var targetPlid = item.getAttribute('data-plid');
        
        // 기본 노출 상태 설정
        var shouldShow = false;

        // 조건 처리
        if (!targetPlid) {
            // 1. targetPlid가 없으면 공통 메뉴이므로 노출
            shouldShow = true;
        } else if (Number(targetPlid) === currentPlid) {
            // 2. targetPlid와 현재 페이지 ID가 같으면 해당 페이지 전용 메뉴이므로 노출
            shouldShow = true;
        } else {
            // 3. 현재 페이지가 40번이 아닐 때
            // 단, 40번이 아닌 페이지들 중에서 targetPlid가 42인 것은 숨김
            if (currentPlid !== 40 && currentPlid !== 42) {
                if (Number(targetPlid) === 42) {
                    shouldShow = false;
                } else {
                    shouldShow = true;
                }
            }
        }

        // 최종 결과 적용
        if (shouldShow) {
            item.style.display = 'block';
        } else {
            item.style.display = 'none';
        }
    });
});

// <![CDATA[
/* =================================================================================
   PART 1. [UI 렌더링] - Liferay Loader와 상관없이 즉시 실행 (순수 JS)
   - 기존 버튼 숨김 (display:none)
   - 새 버튼 생성 및 삽입
   - 화면 크기(Mobile/PC)에 따른 노출 제어
   ================================================================================= */
(function() {
    try {
        // 1-1. 기존 메뉴 버튼 (모바일 햄버거)
        var oldMenuBtn = document.querySelector('button.navbar-toggler[data-target="#navigationCollapse"]');
        
        // 1-2. 기존 검색 버튼 (선택자 수정: 클래스일 경우 . 추가)
        // 태그명이 아니라면 반드시 앞에 점(.)을 붙여야 합니다.
        var searchBtn = document.querySelector('.searchbar-toggler-modal[data-target="#searchbarCollapse"]') || 
                        document.querySelector('button[data-target="#searchbarCollapse"]'); 

        // ==========================================
        // 2. 반응형 핸들러 함수 정의 (하나로 통합)
        // ==========================================
        var newMenuBtn, newLangBtn;

        var handleScreenChange = function(e) {
            var isMobile = e.matches; // 991px 이하일 때 true

            // A. 메뉴 버튼 제어
            if (oldMenuBtn && newMenuBtn) {
                if (isMobile) {
                    oldMenuBtn.style.display = 'none';
                    newMenuBtn.style.display = 'block';
                } else {
                    oldMenuBtn.style.display = '';
                    newMenuBtn.style.display = 'none';
                }
            }

            // B. 언어 버튼 제어
            if (newLangBtn) {
                if (isMobile) {
                    newLangBtn.style.display = 'block';
                } else {
                    newLangBtn.style.display = 'none';
                }
            }
        };

        // ==========================================
        // 3. 버튼 생성 및 삽입 로직
        // ==========================================

        // 3-1. 메뉴 버튼 처리
        if (oldMenuBtn && !document.querySelector('.mobile-injected-btn')) {
            newMenuBtn = document.createElement('button');
            newMenuBtn.setAttribute('type', 'button');
            newMenuBtn.className = 'btn btn_menuToggle mobile-injected-btn';
            newMenuBtn.style.display = 'none';
            
            // 기존 버튼 뒤에 삽입
            oldMenuBtn.parentNode.insertBefore(newMenuBtn, oldMenuBtn.nextSibling);
        }

        // 3-2. 언어(검색) 버튼 처리
        if (searchBtn && !document.querySelector('.mobile-searched-btn')) {
            newLangBtn = document.createElement('a');
            newLangBtn.setAttribute('href', '/web/www');
            newLangBtn.className = 'lang_current mobile-searched-btn';
            newLangBtn.style.width = 'max-content';
            newLangBtn.textContent = '한국어'; // (옵션) 텍스트가 필요하다면 추가
            newLangBtn.style.display = 'none';

            // [핵심 수정] 기존 버튼 '앞'에 삽입
            // newLangBtn은 부모가 없으므로 searchBtn의 부모를 호출해야 함
            searchBtn.parentNode.insertBefore(newLangBtn, searchBtn);
        }

        // ==========================================
        // 4. 리스너 등록 및 초기 실행
        // ==========================================
        var mql = window.matchMedia("screen and (max-width: 991px)");
        if (mql.addEventListener) {
            mql.addEventListener("change", handleScreenChange);
        } else {
            mql.addListener(handleScreenChange);
        }
        
        // 초기 상태 반영
        handleScreenChange(mql);

    } catch (e) {
        console.error("HYU Mobile Script Error:", e);
    }
})();

/* =================================================================================
   PART 2. [기능 연결] - AUI 모듈 로드 후 실행
   - 메뉴 토글 기능 구현
   - 이벤트 위임(Delegate)으로 동적 생성된 버튼 클릭 감지
   ================================================================================= */
AUI().ready('aui-base', 'node-event-delegate', function(A) {
    
    var navigation = A.one('#muaf_navbar_com_liferay_site_navigation_menu_web_portlet_SiteNavigationMenuPortlet');
    var menuPopup = A.one('.menu-popup-wrap');
    var menuClose = A.one('.btn_menuClose');
    var mobileWrap = A.one('.navbar-collapse.navbar-collapse-slide')
    var mobileMenuWrap = A.one('.m-hyu-menu-depth-content-wrap');
    var isMobileView = window.matchMedia("(max-width: 991px)").matches;

    // ---------------------------------------------------------------
    // 1. 메뉴 토글 함수 정의
    // ---------------------------------------------------------------
    var toggleMenu = function(open) {
        if (!menuPopup) return;
        
        // PC버튼, 모바일버튼 모두 찾아서 스타일 동기화
        var pcToggles = A.all('.btn_menuToggle.pc-btn');
        var moToggles = A.all('.btn_menuToggle.mobile-injected-btn');
        
        if (open) {
            // 1. 공통 동작 (메뉴 팝업 표시, 초기화, 스크롤 잠금)
            menuPopup.addClass('show');
            initializeMenuState();
            document.body.style.overflow = 'hidden';
        
            // 2. 해상도에 따른 버튼(Toggle) 분기 처리
            if (isMobileView) {
                // 991px 이하: 모바일 토글 버튼에만 class 추가
                moToggles.addClass('open');
                mobileWrap.addClass('show');
                
                // (선택사항) PC 토글이 열려있었다면 닫아줌 (안전장치)
                pcToggles.removeClass('open'); 
            } else {
                // 992px 이상: PC 토글 버튼에만 class 추가
                pcToggles.addClass('open');
                
                // (선택사항) 모바일 토글이 열려있었다면 닫아줌
                moToggles.removeClass('open');
            }
        
        } else {
            // 1. 공통 동작 (메뉴 팝업 숨김, 리셋, 스크롤 복구)
            menuPopup.removeClass('show');
            resetMenuState();
            document.body.style.overflow = '';
        
            // 2. 닫을 때는 해상도 구분 없이 모두 removeClass 하는 것이 UI 버그 방지에 좋습니다.
            //    (화면 크기를 조절하며 테스트할 때 꼬임 방지)
            moToggles.removeClass('open');
            pcToggles.removeClass('open');
            mobileWrap.removeClass('show');
        }
    };

    // ---------------------------------------------------------------
    // 2. 이벤트 바인딩 (Delegate 사용 - 중요!)
    // 문서 전체(Body)에서 .btn_menuToggle 클릭을 감지하므로, 
    // 위에서 순수 JS로 만든 버튼도 여기서 작동됩니다.
    // ---------------------------------------------------------------
    A.getBody().delegate('click', function(e) {
        e.preventDefault();
        e.stopPropagation();

        // 현재 팝업이 열려있는지 확인
        var isOpened = menuPopup.hasClass('show');
        
        // 상태 반전하여 실행
        toggleMenu(!isOpened);
        
    }, '.btn_menuToggle'); // <- 이 클래스를 가진 놈이 클릭되면 실행


    // 닫기 버튼
    if (menuClose) {
        menuClose.on('click', function(e) {
            e.preventDefault();
            toggleMenu(false);
        });
    }

    // 언어 선택
    var languageSelect = A.one('#item-lang');
    if (languageSelect) {
        languageSelect.on('change', function() {
            var val = languageSelect.val();
            if (val) window.location.href = val;
        });
    }

    // ---------------------------------------------------------------
    // 3. PC / Mobile 뎁스 네비게이션 로직 (기존 유지)
    // ---------------------------------------------------------------
    
    // 상태 초기화 함수
    var resetMenuState = function() {
        if (!navigation) return;
        navigation.all('.nav-item-tab').removeClass('hover-active');
        navigation.all('.hyu-menu-content').removeClass('hover-show');
        navigation.all('.depth2-item').removeClass('hover-active');
        navigation.all('.hyu-menu-depth3').removeClass('hover-show');
        
        if (mobileMenuWrap) {
            mobileMenuWrap.all('.hyu-menu-content').setStyle('display', 'none');
            mobileMenuWrap.all('.hyu-menu-depth3-wrap').setStyle('display', 'none');
            mobileMenuWrap.all('li').removeClass('active');
        }
    };

    // 초기 활성화 함수
    var initializeMenuState = function() {
        resetMenuState();
        if (!navigation) return;
        
        var activeTab = navigation.one('.nav-item-tab.active') || navigation.one('.nav-item-tab:first-child');
        if (activeTab && window.innerWidth >= 992) {
            activeTab.addClass('hover-active');
            var targetId = activeTab.getAttribute('data-target-id');
            var activeContent = navigation.one('#' + targetId);
            if (activeContent) {
                activeContent.addClass('hover-show');
                var activeDepth2 = activeContent.one('.depth2-item.active') || activeContent.one('.depth2-item:first-child');
                if (activeDepth2) {
                    activeDepth2.addClass('hover-active');
                    var target3Id = activeDepth2.getAttribute('data-target-id');
                    var depth3 = activeContent.one('#' + target3Id);
                    if (depth3) depth3.addClass('hover-show');
                }
            }
        }
    };

    // [PC] 마우스 오버
    if (navigation) {
        navigation.all('.nav-item-tab').on('mouseenter', function(e) {
            if (window.innerWidth < 992) return;
            e.preventDefault();
            var targetId = e.currentTarget.getAttribute('data-target-id');
            
            navigation.all('.nav-item-tab').removeClass('hover-active');
            e.currentTarget.addClass('hover-active');
            
            navigation.all('.hyu-menu-content').removeClass('hover-show');
            var activeContent = navigation.one('#' + targetId);
            if (activeContent) {
                activeContent.addClass('hover-show');
                var f2 = activeContent.one('.depth2-item:first-child');
                if (f2) {
                    activeContent.all('.depth2-item').removeClass('hover-active');
                    activeContent.all('.hyu-menu-depth3').removeClass('hover-show');
                    f2.addClass('hover-active');
                    var t3 = f2.getAttribute('data-target-id');
                    var d3 = activeContent.one('#' + t3);
                    if (d3) d3.addClass('hover-show');
                }
            }
        });

        navigation.all('.depth2-item').on('mouseenter', function(e) {
            if (window.innerWidth < 992) return;
            var t3 = e.currentTarget.getAttribute('data-target-id');
            var p = e.currentTarget.ancestor('.hyu-menu-content');
            if (p) {
                p.all('.depth2-item').removeClass('hover-active');
                e.currentTarget.addClass('hover-active');
                p.all('.hyu-menu-depth3').removeClass('hover-show');
                var d3 = p.one('#' + t3);
                if (d3) d3.addClass('hover-show');
            }
        });
    }

    // [Mobile] 아코디언 클릭
    if (mobileMenuWrap) {
        // 모바일 1Depth 클릭
        mobileMenuWrap.delegate('click', function(e) {
            e.preventDefault();
            var li = e.currentTarget.ancestor('li');
            var content = li.one('.hyu-menu-content');
            if (content) {
                var isVisible = (content.getStyle('display') === 'block');
                mobileMenuWrap.all('.hyu-menu-content').setStyle('display', 'none');
                mobileMenuWrap.all('.nav-item-tab').removeClass('active');
                if (!isVisible) {
                    content.setStyle('display', 'block');
                    li.addClass('active');
                }
            }
        }, '.depth1-link');

        // 모바일 2Depth 클릭
        mobileMenuWrap.delegate('click', function(e) {
            var cur = e.currentTarget;
            var hasChildren = (cur.getAttribute('data-has-children') === 'true');

            if (hasChildren) {
                e.preventDefault();
                var li = cur.ancestor('li');
                var content = li.one('.hyu-menu-depth3-wrap');
                if (content) {
                    var isVisible = (content.getStyle('display') === 'block');
                    cur.ancestor('ul').all('.hyu-menu-depth3-wrap').setStyle('display', 'none');
                    cur.ancestor('ul').all('li').removeClass('active');
                    if (!isVisible) {
                        content.setStyle('display', 'block');
                        li.addClass('active');
                    }
                }
            }
        }, '.depth2-link');
    }
    
    // 페이지 로드 시 스크롤바 초기화 (안전장치)
    document.body.style.overflow = '';
});
